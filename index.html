// 프론트엔드의 fetch 부분 수정 async function fetchAll(lat, lng) { // 사용자가 설문에서 선택한 데이터까지 몽땅 백엔드로 넘깁니다. const query = `lat=${lat}&lng=${lng}&lcls=${userPrefs.lcls}&mcls=${userPrefs.mcls}&budget=${userPrefs.budget}`; try { const res = await fetch(`/api/analyze?${query}`); const json = await res.json(); if(json.success) return json.data; else throw new Error("API Error"); } catch (e) { console.error(e); // 에러 시 임시 더미 데이터 (테스트용) return { bizCount:15, monthlyRent:250, targetSales:2500, customersNeededPerDay:185, score:75, analysis:{ summary:"분석 지연", actionItems:["네트워크 확인 필요"] } }; } } // 리포트 렌더링 함수 수정 (팩트 폭격 + 행동 지침 추가) function openReport() { const modal = document.getElementById('reportModal'); const content = document.getElementById('reportContent'); // 백엔드에서 받은 든든한 데이터 const d = analysisData; content.innerHTML = `
상권레이더 AI 컨설팅 결과
"${d.analysis.summary}"
AI 생존 예측 점수 ${d.score}점
🔥 월세 감당을 위한 생존 마지노선

추정 월세 ${d.monthlyRent.toLocaleString()}만 원을 내고 망하지 않으려면, 최소 아래의 목표를 달성해야 합니다.

필요 월 매출 ${d.targetSales.toLocaleString()}만 원
일일 필요 결제자 수 ${d.customersNeededPerDay}명 (객단가 ${d.unitPrice.toLocaleString()}원 기준)

    ${d.analysis.actionItems.map(item => `
  • ${item}
    • `).join('')}
`; modal.classList.add('show'); }
